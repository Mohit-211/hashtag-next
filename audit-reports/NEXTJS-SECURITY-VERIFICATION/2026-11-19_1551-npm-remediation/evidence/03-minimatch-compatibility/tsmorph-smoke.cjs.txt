const dir = process.argv[2];
const { Project, SyntaxKind } = require(dir + "/node_modules/ts-morph");
const path = require("path"); const { createRequire } = require("module"); const req = createRequire(path.join(dir, "node_modules/@ts-morph/common/package.json")); const mmPath = req.resolve("minimatch"); const mmDir = mmPath.slice(0, mmPath.lastIndexOf("minimatch") + "minimatch".length); const mm = require(path.join(mmDir, "package.json")).version + " @ " + path.relative(dir, mmDir);
// 1) What shadcn does: in-memory project, createSourceFile, AST edits, print
const p = new Project({ useInMemoryFileSystem: true, compilerOptions: { allowJs: true } });
const sf = p.createSourceFile("components/ui/button.tsx",
  'import * as React from "react"\nimport { cn } from "@/lib/utils"\nexport function Button(){ return <button className={cn("a")} /> }\n', { overwrite: true });
sf.getImportDeclarations().find(d => d.getModuleSpecifierValue() === "@/lib/utils").setModuleSpecifier("@/components/lib/utils");
const jsx = sf.getDescendantsOfKind(SyntaxKind.JsxAttribute).length;
// 2) Glob paths that DO go through minimatch (getPathMatchesPattern)
const fs = p.getFileSystem();
fs.writeFileSync("/src/a.ts", "export const a=1"); fs.writeFileSync("/src/b/c.tsx", "export const c=1");
fs.writeFileSync("/src/node_modules/x.ts", "x"); fs.writeFileSync("/src/.hidden/h.ts", "h");
const added = p.addSourceFilesAtPaths(["/src/**/*.{ts,tsx}", "!/src/node_modules/**"]).map(f => f.getFilePath()).sort();
const filtered = p.getSourceFiles("**/*.tsx").map(f => f.getFilePath()).sort();
console.log(JSON.stringify({ minimatch: mm, printed: sf.getFullText(), jsxAttrs: jsx, added, filtered }));

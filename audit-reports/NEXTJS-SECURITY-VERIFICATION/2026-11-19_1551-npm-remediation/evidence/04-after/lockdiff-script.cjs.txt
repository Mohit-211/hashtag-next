const fs = require("fs");
const [, , beforeF, afterF] = process.argv;
const A = JSON.parse(fs.readFileSync(beforeF, "utf8")), B = JSON.parse(fs.readFileSync(afterF, "utf8"));
const pa = A.packages, pb = B.packages, out = [];
for (const k of new Set([...Object.keys(pa), ...Object.keys(pb)])) {
  const x = pa[k], y = pb[k];
  if (!x) out.push(`+ ${k}@${y.version}${y.inBundle ? " [inBundle]" : ""}${y.optional ? " [optional]" : ""}${y.dev ? " [dev]" : ""}`);
  else if (!y) out.push(`- ${k}@${x.version}`);
  else if (JSON.stringify(x) !== JSON.stringify(y)) {
    const keys = [...new Set([...Object.keys(x), ...Object.keys(y)])].filter(f => JSON.stringify(x[f]) !== JSON.stringify(y[f]));
    out.push(`~ ${k}: ${x.version} -> ${y.version} (fields changed: ${keys.join(", ")})`);
  }
}
const top = Object.keys(A).filter(k => k !== "packages" && JSON.stringify(A[k]) !== JSON.stringify(B[k]));
console.log(out.join("\n") || "(no package entry changes)");
console.log("top-level lock fields changed:", top.length ? top.join(", ") : "none");
console.log("entries before:", Object.keys(pa).length, "after:", Object.keys(pb).length);
